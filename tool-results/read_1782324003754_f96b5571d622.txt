   115→   115→function round2(n: number): number {
   116→   116→  return Math.round(n * 100) / 100;
   117→   117→}
   118→   118→
   119→   119→function matchPriceItem(
   120→   120→  items: PriceItemRecord[],
   121→   121→  system: string,
   122→   122→  category: string,
   123→   123→  keywords: string[]
   124→   124→): PriceItemRecord | null {
   125→   125→  const lowered = keywords.map((k) => k.toLowerCase());
   126→   126→  return (
   127→   127→    items.find((item) => {
   128→   128→      if (item.system !== system) return false;
   129→   129→      if (category && item.category !== category) return false;
   130→   130→      const desc = item.description.toLowerCase();
   131→   131→      return lowered.some((kw) => desc.includes(kw));
   132→   132→    }) ?? null
   133→   133→  );
   134→   134→}
   135→   135→
   136→   136→function makeLineItem(
   137→   137→  idx: number,
   138→   138→  systemPrefix: string,
   139→   139→  system: SystemName,
   140→   140→  code: string,
   141→   141→  description: string,
   142→   142→  unit: string,
   143→   143→  quantity: number,
   144→   144→  unitCost: number,
   145→   145→  category: string = "Equipo",
   146→   146→  isEstimated = false
   147→   147→): LineItem {
   148→   148→  return {
   149→   149→    partida: `${systemPrefix}.${String(idx).padStart(2, "0")}`,
   150→   150→    code,
   151→   151→    description,
   152→   152→    unit,
   153→   153→    quantity: round2(quantity),
   154→   154→    unitCost: round2(unitCost),
   155→   155→    totalAmount: round2(quantity * unitCost),
   156→   156→    system,
   157→   157→    category,
   158→   158→    isEstimated,
   159→   159→  };
   160→   160→}
   161→   161→
   162→   162→function sumLineItems(items: LineItem[]): { materials: number; labor: number; engineering: number; services: number } {
   163→   163→  let materials = 0;
   164→   164→  let labor = 0;
   165→   165→  let engineering = 0;
   166→   166→  let services = 0;
   167→   167→  for (const item of items) {
   168→   168→    const amt = item.totalAmount;
   169→   169→    if (item.category === "Mano de Obra") labor += amt;
   170→   170→    else if (item.category === "Servicio") services += amt;
   171→   171→    else if (item.category === "Ingeniería") engineering += amt;
   172→   172→    else materials += amt;
   173→   173→  }
   174→   174→  return { materials: round2(materials), labor: round2(labor), engineering: round2(engineering), services: round2(services) };
   175→   175→}

   179→   179→export function calculateCCTV(
   180→   180→  config: CctvConfig,
   181→   181→  factors: EstimateFactors,
   182→   182→  priceItems: PriceItemRecord[]
   183→   183→): SystemBreakdown {
   184→   184→  const items: LineItem[] = [];
   185→   185→  let idx = 1;
   186→   186→
   187→   187→  // Total camera count
   188→   188→  const totalCameras = config.cameras.reduce((sum, c) => sum + c.qty, 0);
   189→   189→
   190→   190→  // Cable calculation for cameras
   191→   191→  const cableLength = round2(
   192→   192→    (config.avgDistanceMeters + factors.verticalDrop + factors.rackAllowance) * totalCameras * (1 + factors.wasteFactorCable)
   193→   193→  );
   194→   194→
   195→   195→  // Conduit calculation for cameras
   196→   196→  const conduitLength = round2(config.avgDistanceMeters * totalCameras * (1 + factors.wasteFactorConduit));
   197→   197→
   198→   198→  // --- Cameras ---
   199→   199→  for (const cam of config.cameras) {
   200→   200→    const typeLower = cam.type.toLowerCase();
   201→   201→    const keywords: string[] = [];
   202→   202→    if (typeLower.includes("bullet")) keywords.push("bullet");
   203→   203→    else if (typeLower.includes("domo") || typeLower.includes("dome")) keywords.push("domo");
   204→   204→    else if (typeLower.includes("ptz")) keywords.push("ptz");
   205→   205→    else if (typeLower.includes("fisheye")) keywords.push("fisheye");
   206→   206→
   207→   207→    const matched = matchPriceItem(priceItems, "CCTV", "Equipo", keywords);
   208→   208→    const unitCost = matched?.unitCost ?? getEstimatedCost("cctv_camera_" + typeLower);
   209→   209→
   210→   210→    items.push(
   211→   211→      makeLineItem(
   212→   212→        idx++, "5.7.3", "CCTV", `CCTV-CAM-${String(idx).padStart(3, "0")}`,
   213→   213→        `Cámara IP ${cam.type} ${cam.hasPoE ? "PoE" : ""} exterior 5MP`,
   214→   214→        "pza", cam.qty, unitCost, "Equipo", !matched
   215→   215→      )
   216→   216→    );
   217→   217→  }
   218→   218→
   219→   219→  // --- NVR ---
   220→   220→  const nvrMatch = matchPriceItem(priceItems, "CCTV", "Equipo", ["NVR", "grabador"]);
   221→   221→  const nvrCost = nvrMatch?.unitCost ?? getEstimatedCost("cctv_nvr");
   222→   222→  items.push(
   223→   223→    makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-NVR-001", `NVR ${config.nvr.bays} bahías`, "pza", config.nvr.qty, nvrCost, "Equipo", !nvrMatch)
   224→   224→  );
   225→   225→
   226→   226→  // --- Storage Disks ---
   227→   227→  const totalDisks = config.nvr.bays * config.nvr.disksPerBay;
   228→   228→  const diskMatch = matchPriceItem(priceItems, "CCTV", "Equipo", ["disco", "disco duro", "HDD", "almacenamiento"]);
   229→   229→  const diskCost = diskMatch?.unitCost ?? getEstimatedCost("cctv_disk");
   230→   230→  if (totalDisks > 0) {
   231→   231→    items.push(
   232→   232→      makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-DISK-001", `Disco duro ${config.nvr.storageTB}TB para NVR`, "pza", totalDisks, diskCost, "Equipo", !diskMatch)
   233→   233→    );
   234→   234→  }
   235→   235→
   236→   236→  // --- Licenses ---
   237→   237→  if (config.licenses > 0) {
   238→   238→    const licMatch = matchPriceItem(priceItems, "CCTV", "Equipo", ["licencia", "license"]);
   239→   239→    const licCost = licMatch?.unitCost ?? getEstimatedCost("cctv_license");
   240→   240→    items.push(
   241→   241→      makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-LIC-001", "Licencia de cámara IP adicional", "pza", config.licenses, licCost, "Equipo", !licMatch)
   242→   242→    );
   243→   243→  }
   244→   244→
   245→   245→  // --- Cable UTP ---
   246→   246→  const cableMatch = matchPriceItem(priceItems, "CABLEADO", "Equipo", ["UTP", "cable UTP", "cable Cat"]);
   247→   247→  const cableCost = cableMatch?.unitCost ?? getEstimatedCost("cable_utp");
   248→   248→  items.push(
   249→   249→    makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-CAB-001", "Cable UTP Cat6 para CCTV", "ml", cableLength, cableCost, "Equipo", !cableMatch)
   250→   250→  );
   251→   251→
   252→   252→  // --- Conduit ---
   253→   253→  const conduitMatch = matchPriceItem(priceItems, "CANALIZACION", "Equipo", ["conduit", "tubo"]);
   254→   254→  const conduitCost = conduitMatch?.unitCost ?? getEstimatedCost("conduit");
   255→   255→  items.push(
   256→   256→    makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-CON-001", "Conduit 3/4\" para CCTV", "ml", conduitLength, conduitCost, "Equipo", !conduitMatch)
   257→   257→  );
   258→   258→
   259→   259→  // --- Conduit fittings (codo, cople) ---
   260→   260→  const fittingCount = Math.ceil(conduitLength / 3);
   261→   261→  const codoMatch = matchPriceItem(priceItems, "CANALIZACION", "Accesorio", ["codo"]);
   262→   262→  const codoCost = codoMatch?.unitCost ?? getEstimatedCost("codo");
   263→   263→  if (fittingCount > 0) {
   264→   264→    items.push(
   265→   265→      makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-CODO-001", "Codo conduit 3/4\" 90°", "pza", fittingCount, codoCost, "Accesorio", !codoMatch)
   266→   266→    );
   267→   267→  }
   268→   268→
   269→   269→  const copleMatch = matchPriceItem(priceItems, "CANALIZACION", "Accesorio", ["cople"]);
   270→   270→  const copleCost = copleMatch?.unitCost ?? getEstimatedCost("cople");
   271→   271→  if (fittingCount > 0) {
   272→   272→    items.push(
   273→   273→      makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-COPL-001", "Cople conduit 3/4\"", "pza", fittingCount, copleCost, "Accesorio", !copleMatch)
   274→   274→    );
   275→   275→  }
   276→   276→
   277→   277→  // --- Labor ---
   278→   278→  const laborMatch = matchPriceItem(priceItems, "CCTV", "Mano de Obra", ["mano de obra", "instalación"]);
   279→   279→  const laborCost = laborMatch?.unitCost ?? getEstimatedCost("labor_cctv");
   280→   280→  items.push(
   281→   281→    makeLineItem(idx++, "5.7.3", "CCTV", "CCTV-MO-001", "Instalación de sistema CCTV (mano de obra)", "lote", 1, laborCost * totalCameras, "Mano de Obra", !laborMatch)
   282→   282→  );
   283→   283→
   284→   284→  const sums = sumLineItems(items);
   285→   285→  return { ...sums, lineItems: items };
   286→   286→}

   398→   398→export function calculatePaging(
   399→   399→  config: PagingConfig,
   400→   400→  factors: EstimateFactors,
   401→   401→  priceItems: PriceItemRecord[]
   402→   402→): SystemBreakdown {
   403→   403→  const items: LineItem[] = [];
   404→   404→  let idx = 1;
   405→   405→
   406→   406→  const totalSpeakers = config.speakers.reduce((s, sp) => s + sp.qty, 0) + config.bluetoothSpeakers;
   407→   407→  const totalDevices = totalSpeakers + config.amplifiers.qty + config.gateways;
   408→   408→
   409→   409→  // Cable
   410→   410→  const cableLength = round2(
   411→   411→    (config.avgDistanceMeters + factors.verticalDrop + factors.rackAllowance) * totalDevices * (1 + factors.wasteFactorCable)
   412→   412→  );
   413→   413→
   414→   414→  // Conduit
   415→   415→  const conduitLength = round2(config.avgDistanceMeters * totalDevices * (1 + factors.wasteFactorConduit));
   416→   416→
   417→   417→  // --- Speakers ---
   418→   418→  for (const sp of config.speakers) {
   419→   419→    const typeLower = sp.type.toLowerCase();
   420→   420→    const keywords: string[] = [];
   421→   421→    if (typeLower.includes("exterior")) keywords.push("exterior", "bafle", "protección");
   422→   422→    else if (typeLower.includes("colgante")) keywords.push("colgante", "techo");

   680→   680→export function runCalculation(params: {
   681→   681→  cctvConfig?: CctvConfig;
   682→   682→  accessConfig?: AccessConfig;
   683→   683→  pagingConfig?: PagingConfig;
   684→   684→  fireConfig?: FireConfig;
   685→   685→  factors: EstimateFactors;
   686→   686→  priceItems: PriceItemRecord[];
   687→   687→}): CalculationResult {
   688→   688→  const { cctvConfig, accessConfig, pagingConfig, fireConfig, factors, priceItems } = params;
   689→   689→
   690→   690→  const systems: Record<string, SystemBreakdown> = {};
   691→   691→  const allLineItems: LineItem[] = [];
   692→   692→
   693→   693→  if (cctvConfig && cctvConfig.cameras?.length > 0) {
   694→   694→    systems["CCTV"] = calculateCCTV(cctvConfig, factors, priceItems);
   695→   695→    allLineItems.push(...systems["CCTV"].lineItems);
   696→   696→  }
   697→   697→
   698→   698→  if (accessConfig && accessConfig.doors > 0) {
   699→   699→    systems["ACCESO"] = calculateAccess(accessConfig, factors, priceItems);
   700→   700→    allLineItems.push(...systems["ACCESO"].lineItems);
   701→   701→  }
   702→   702→
   703→   703→  if (pagingConfig && (pagingConfig.speakers?.length > 0 || pagingConfig.bluetoothSpeakers > 0)) {
   704→   704→    systems["VOCEO"] = calculatePaging(pagingConfig, factors, priceItems);
   705→   705→    allLineItems.push(...systems["VOCEO"].lineItems);
   706→   706→  }
   707→   707→
   708→   708→  if (fireConfig && (fireConfig.smokeDetectors > 0 || fireConfig.heatDetectors > 0 || fireConfig.manualStations > 0)) {
   709→   709→    systems["INCENDIO"] = calculateFire(fireConfig, factors, priceItems);
   710→   710→    allLineItems.push(...systems["INCENDIO"].lineItems);
   711→   711→  }
   712→   712→
   713→   713→  // Add general/service items from price items
   714→   714→  const serviceItems = priceItems.filter(
   715→   715→    (pi) => pi.system === "GENERAL" && pi.category === "Servicio" && pi.active
   716→   716→  );
   717→   717→  const generalLineItems: LineItem[] = [];
   718→   718→  serviceItems.forEach((si, i) => {
   719→   719→    generalLineItems.push(
   720→   720→      makeLineItem(
   721→   721→        i + 1, "5.7.0", "GENERAL", `SRV-${String(i + 1).padStart(3, "0")}`,
   722→   722→        si.description, si.unit, 1, si.unitCost, "Servicio"
   723→   723→      )
   724→   724→    );
   725→   725→  });
   726→   726→  if (generalLineItems.length > 0) {
   727→   727→    systems["GENERAL"] = { ...sumLineItems(generalLineItems), lineItems: generalLineItems };
   728→   728→    allLineItems.push(...generalLineItems);
   729→   729→  }
   730→   730→
   731→   731→  const subtotalMaterials = round2(allLineItems.filter((i) => i.category === "Equipo" || i.category === "Accesorio" || i.category === "Consumible").reduce((s, i) => s + i.totalAmount, 0));
   732→   732→  const subtotalLabor = round2(allLineItems.filter((i) => i.category === "Mano de Obra").reduce((s, i) => s + i.totalAmount, 0));
   733→   733→  const subtotalEngineering = round2(allLineItems.filter((i) => i.category === "Ingeniería").reduce((s, i) => s + i.totalAmount, 0));
   734→   734→  const subtotalServices = round2(allLineItems.filter((i) => i.category === "Servicio").reduce((s, i) => s + i.totalAmount, 0));
   735→   735→
   736→   736→  const subtotalDirect = round2(subtotalMaterials + subtotalLabor + subtotalEngineering + subtotalServices);
   737→   737→  const subtotalIndirects = round2(subtotalDirect * factors.indirectFactor);
   738→   738→  const subtotalUtility = round2((subtotalDirect + subtotalIndirects) * factors.utilityFactor);
   739→   739→  const grandTotal = round2(subtotalDirect + subtotalIndirects + subtotalUtility);
   740→   740→
   741→   741→  return {
   742→   742→    systems,
   743→   743→    subtotalMaterials,
   744→   744→    subtotalLabor,
   745→   745→    subtotalEngineering,
   746→   746→    subtotalServices,
   747→   747→    subtotalDirect,
   748→   748→    subtotalIndirects,
   749→   749→    subtotalUtility,
   750→   750→    grandTotal,
   751→   751→    lineItems: allLineItems,
   752→   752→  };
   753→   753→}
